Sucursales:

LOAD
    If(IsNum(Sucursal),
        Right('00' & Num#(Sucursal), 2),
        Sucursal
    ) as Sucursal,
    If(Sucursal = 'E1', 'MELI Tesi',
    If(Sucursal = 'E2', 'MELI Pueblo',
    If(Sucursal = 'WE1', 'WEB Tesi',
    If(Sucursal = 'WE2', 'WEB Pueblo',
    If(Sucursal = 'FK1', 'WEB Freekick',
        nomSucursal))))) as nomSucursal
WHERE IsNum(Sucursal) 
   OR Match(Sucursal, 'E1','E2','WE1','WE2','FK1');

SQL SELECT
    Sucursal,
    nomSucursal
FROM db_Cegid.dbo.Sucursales
WHERE sucvta = 'S'
AND viewsuc = 'S';
